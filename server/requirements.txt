# Markdown → HTML rendering for the workspace import path.

# PDF editor parse + export pipeline lives on PyMuPDF directly. No
# DOCX / PPTX / scan-OCR conversion happens server-side any more — the
# workspace folder is the source of truth and external formats are
# expected to land there as `.md` or `.pdf` directly.
pymupdf>=1.24.0

# Excel editor parse + export pipeline. openpyxl handles read/modify/write
# of .xlsx and .xlsm with full style fidelity. We deliberately do not pull
# in calamine here: it can't write, and the read speed gap only matters for
# multi-MB sheets which the editor isn't optimised for yet.
openpyxl>=3.1.5

# Cross-platform OS trash for the doc_delete fallback used by the browser
# dev surface. On macOS / Windows it talks to the native recycle bin; the
# Tauri app uses the Rust `trash` crate directly and never touches this.
send2trash>=1.8.3

# Fast JSON encoder/decoder. Used by the parsed-workbook LRU in
# services/excel_workbook.py: cache values are stored as JSON-encoded
# bytes so each hit returns an isolated dict via a fresh decode rather
# than copy.deepcopy. orjson roughly halves both the encode and decode
# cost vs stdlib json on the multi-MB DTOs the cache holds (measured:
# 1180 ms -> 488 ms for the 8 MB fixture round-trip).
orjson>=3.10.0

# `doxmind` CLI (ADR 0010). Typer builds the command surface on top of Click;
# the CLI imports the core facade directly and runs without the desktop app.
typer==0.26.7

# `doxmind-mcp` server (ADR 0010). The official MCP SDK; FastMCP exposes the
# workspace to external agents over stdio. The local package is named
# `doxmind_mcp` so it does not shadow this `mcp` import. Pinned: newer builds
# changed FastMCP's parameter-annotation handling, so an unpinned `>=` resolved
# a version on CI that rejected the tool signatures.
mcp==1.27.2
